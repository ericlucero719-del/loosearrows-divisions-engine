// scripts/buildShopifyDivisionCatalogs.ts
import fetch from "node-fetch";
import { registry } from "../src/core/engine";

const SHOPIFY_STORE_DOMAIN = process.env.SHOPIFY_STORE_DOMAIN as string; // e.g. "your-store.myshopify.com"
const SHOPIFY_ACCESS_TOKEN = process.env.SHOPIFY_ACCESS_TOKEN as string;

if (!SHOPIFY_STORE_DOMAIN || !SHOPIFY_ACCESS_TOKEN) {
  throw new Error("Missing SHOPIFY_STORE_DOMAIN or SHOPIFY_ACCESS_TOKEN env vars");
}

type DivisionConfig = {
  id: number;
  name: string;
  collectionTitle: string;
  collectionHandle: string;
};

const divisions: DivisionConfig[] = [
  { id: 1, name: "Product Intake & Pricing", collectionTitle: "Division 1 Catalog", collectionHandle: "division-1" },
  { id: 2, name: "Contract Alignment", collectionTitle: "Division 2 Catalog", collectionHandle: "division-2" },
  { id: 3, name: "Requests & Work Orders", collectionTitle: "Division 3 Catalog", collectionHandle: "division-3" },
  { id: 4, name: "Inventory & Assets", collectionTitle: "Division 4 Catalog", collectionHandle: "division-4" },
  { id: 5, name: "Logistics & Fulfillment", collectionTitle: "Division 5 Catalog", collectionHandle: "division-5" },
  { id: 6, name: "Compliance & Documentation", collectionTitle: "Division 6 Catalog", collectionHandle: "division-6" },
  { id: 7, name: "Vendors & Partners", collectionTitle: "Division 7 Catalog", collectionHandle: "division-7" },
  { id: 8, name: "Agencies & Customers", collectionTitle: "Division 8 Catalog", collectionHandle: "division-8" },
  { id: 9, name: "Financials", collectionTitle: "Division 9 Catalog", collectionHandle: "division-9" },
  { id: 10, name: "System Intelligence", collectionTitle: "Division 10 Catalog", collectionHandle: "division-10" }
];

async function shopifyRequest(path: string, method: string, body?: any) {
  const url = `https://${SHOPIFY_STORE_DOMAIN}/admin/api/2024-01${path}`;
  const res = await fetch(url, {
    method,
    headers: {
      "Content-Type": "application/json",
      "X-Shopify-Access-Token": SHOPIFY_ACCESS_TOKEN
    },
    body: body ? JSON.stringify(body) : undefined
  });

  if (!res.ok) {
    const text = await res.text();
    throw new Error(`Shopify error ${res.status}: ${text}`);
  }

  return res.json();
}

async function createCollection(title: string, handle: string) {
  const body = {
    custom_collection: {
      title,
      handle,
      published: true
    }
  };

  const data = await shopifyRequest("/custom_collections.json", "POST", body);
  return data.custom_collection.id as number;
}

async function createProductForDivision(product: any, division: DivisionConfig) {
  const body = {
    product: {
      title: product.productName || product.sku,
      body_html: product.description || "",
      vendor: product.brand || "LooseArrows",
      product_type: product.category || division.name,
      tags: [
        `division:${division.id}`,
        `division-name:${division.name}`,
        product.naics ? `naics:${product.naics}` : null,
        product.clin ? `clin:${product.clin}` : null
      ].filter(Boolean),
      variants: [
        {
          sku: product.sku,
          price: product.price?.toString() || "0.00"
        }
      ],
      images: product.imageUrl
        ? [
            {
              src: product.imageUrl
            }
          ]
        : []
    }
  };

  const data = await shopifyRequest("/products.json", "POST", body);
  return data.product.id as number;
}

async function addProductToCollection(productId: number, collectionId: number) {
  const body = {
    collect: {
      product_id: productId,
      collection_id: collectionId
    }
  };

  await shopifyRequest("/collects.json", "POST", body);
}

function getProductsForDivision(division: DivisionConfig): any[] {
  // Adjust this filter to match how you tag/segment products per division.
  // For now, we’ll just return all products for every division as a demo.
  return registry.products || [];
}

async function buildDivisionCatalog(division: DivisionConfig) {
  console.log(`\n=== Building Shopify catalog for Division ${division.id}: ${division.name} ===`);

  const collectionId = await createCollection(division.collectionTitle, division.collectionHandle);
  console.log(`Created collection ${division.collectionTitle} (ID: ${collectionId})`);

  const products = getProductsForDivision(division);

  for (const p of products) {
    const productId = await createProductForDivision(p, division);
    await addProductToCollection(productId, collectionId);
    console.log(`Linked product ${p.sku || p.productName} to collection ${division.collectionTitle}`);
  }
}

async function main() {
  if (!registry.products || registry.products.length === 0) {
    console.warn("No products in registry.products. Import products via Division 1 before running this script.");
    return;
  }

  for (const division of divisions) {
    await buildDivisionCatalog(division);
  }

  console.log("\nAll 10 division catalogs built in Shopify.");
}

main().catch(err => {
  console.error("Error building Shopify division catalogs:", err);
  process.exit(1);
});
